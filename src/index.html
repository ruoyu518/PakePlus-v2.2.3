        function loadMediaFile(file) {

            if (!file) return;

            stopCurrentMedia();

            const mediaURL = URL.createObjectURL(file);

            currentMediaSrc = mediaURL;

            if (file.type.startsWith('video/')) {

                currentMediaType = 'video';

                loadVideo(mediaURL);

            } else if (file.type.startsWith('image/')) {

                currentMediaType = 'image';

            } else {

                showStatus('不支持的文件类型，请上传视频或图片', 'error');

                return;

            }

        }

                // 核心优化2：监听canplay事件，确保视频纹理可被渲染

                video.addEventListener('canplay', function onCanPlay() {

                        if (videoTexture) videoTexture.dispose();

                        videoTexture.magFilter = THREE.LinearFilter;

                        sphere.material.map = videoTexture;

                        sphere.material.color.set(0xffffff);

                        sphere.material.wireframe = false;

                        sphere.material.transparent = false;

                        sphere.material.opacity = 1;

                    } catch (textureError) {

                        console.error('视频纹理创建失败:', textureError);

                        loadingOverlay.classList.remove('active');

                        showStatus('视频纹理创建失败，视频编码可能不支持', 'error');

                    }

                }, { once: true });

                // 尝试自动播放以解锁音频，但立即暂停

                const playPromise = video.play();

                if (playPromise !== undefined) {

                    playPromise.then(() => { video.pause(); }).catch(e => { /* 忽略自动播放失败错误 */ });

                }

            });

            });

            });

        }

            }

        }

        function onPointerDown(e) { if (e.isPrimary === false || e.button !== 0) return; isUserInteracting = true; onPointerDownPointerX = e.clientX; onPointerDownPointerY = e.clientY; onPointerDownLon = lon; onPointerDownLat = lat; }

        function onPointerMove(e) { if (e.isPrimary === false || !isUserInteracting || isGyroEnabled) return; const dx = e.clientX - onPointerDownPointerX; const dy = e.clientY - onPointerDownPointerY; const sensitivity = 0.1; const fovSensetivity = camera.fov / 60; lon = (onPointerDownLon - dx * sensitivity * fovSensetivity); lat = (onPointerDownLat + dy * sensitivity * fovSensetivity); }

        function onPointerUp() { isUserInteracting = false; }

        function onDocumentMouseWheel(e) { e.preventDefault(); e.deltaY < 0 ? zoomIn() : zoomOut(); }

        function onTouchStart(e) { if (e.touches.length === 2) { isPinchZooming = true; isUserInteracting = false; const dx = e.touches[0].pageX - e.touches[1].pageX; const dy = e.touches[0].pageY - e.touches[1].pageY; pinchZoomDistance = Math.sqrt(dx * dx + dy * dy); initialFov = camera.fov; } else if (e.touches.length === 1) { isUserInteracting = true; onPointerDownPointerX = e.touches[0].clientX; onPointerDownPointerY = e.touches[0].clientY; onPointerDownLon = lon; onPointerDownLat = lat; } }

        function onTouchMove(e) { if (e.touches.length === 2 && isPinchZooming) { e.preventDefault(); const dx = e.touches[0].pageX - e.touches[1].pageX; const dy = e.touches[0].pageY - e.touches[1].pageY; const currentDistance = Math.sqrt(dx * dx + dy * dy); if (pinchZoomDistance > 0) { const scale = pinchZoomDistance / currentDistance; let newFov = initialFov * scale; newFov = THREE.MathUtils.clamp(newFov, minFov, maxFov); camera.fov = newFov; camera.updateProjectionMatrix(); } } else if (e.touches.length === 1 && isUserInteracting && !isGyroEnabled) { const dx = e.touches[0].clientX - onPointerDownPointerX; const dy = e.touches[0].clientY - onPointerDownPointerY; const sensitivity = 0.25; const fovSensetivity = camera.fov / 60; lon = (onPointerDownLon - dx * sensitivity * fovSensetivity); lat = (onPointerDownLat + dy * sensitivity * fovSensetivity); } }

        function onTouchEnd(e) { if (e.touches.length < 2) isPinchZooming = false; if (e.touches.length === 0) isUserInteracting = false; }

        function zoomIn() { let newFov = camera.fov - 5; newFov = THREE.MathUtils.clamp(newFov, minFov, maxFov); camera.fov = newFov; camera.updateProjectionMatrix(); }

        function zoomOut() { let newFov = camera.fov + 5; newFov = THREE.MathUtils.clamp(newFov, minFov, maxFov); camera.fov = newFov; camera.updateProjectionMatrix(); }

        function resetZoom() { camera.fov = initialFov; camera.updateProjectionMatrix(); }

        function updateCamera() { if (isGyroEnabled) return; lat = Math.max(-85, Math.min(85, lat)); phi = THREE.MathUtils.degToRad(90 - lat); theta = THREE.MathUtils.degToRad(lon); const x = 500 * Math.sin(phi) * Math.cos(theta); const y = 500 * Math.cos(phi); const z = 500 * Math.sin(phi) * Math.sin(theta); camera.lookAt(x, y, z); }

        function toggleGyro() { if (!isGyroEnabled) { if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') { DeviceOrientationEvent.requestPermission().then(response => { if (response === 'granted') enableGyro(); else alert('设备方向权限被拒绝，无法启用陀螺仪'); }).catch(error => console.error('请求陀螺仪权限:', error)); } else { enableGyro(); } } else { disableGyro(); } }

        function enableGyro() { isGyroEnabled = true; gyroBtn.textContent = '陀螺仪 开'; gyroBtn.style.background = 'rgba(157, 78, 221, 0.2)'; gyroBtn.style.borderColor = '#9d4edd'; window.addEventListener('deviceorientation', onDeviceOrientation); }

        function disableGyro() { isGyroEnabled = false; gyroBtn.textContent = '陀螺仪 关'; gyroBtn.style.background = ''; gyroBtn.style.borderColor = ''; window.removeEventListener('deviceorientation', onDeviceOrientation); }

        function onDeviceOrientation(e) { if (!isGyroEnabled) return; const alpha = e.alpha ? THREE.MathUtils.degToRad(e.alpha) : 0; const beta = e.beta ? THREE.MathUtils.degToRad(e.beta) : 0; const gamma = e.gamma ? THREE.MathUtils.degToRad(e.gamma) : 0; camera.rotation.x = -beta; camera.rotation.y = -alpha; camera.rotation.z = gamma; }

        function toggleFullscreen() { if (!document.fullscreenElement) { if (document.documentElement.requestFullscreen) document.documentElement.requestFullscreen(); else if (document.documentElement.mozRequestFullScreen) document.documentElement.mozRequestFullScreen(); else if (document.documentElement.webkitRequestFullscreen) document.documentElement.webkitRequestFullscreen(); else if (document.documentElement.msRequestFullscreen) document.documentElement.msRequestfullscreen(); } else { if (document.exitFullscreen) document.exitFullscreen(); else if (document.mozCancelFullScreen) document.mozCancelFullScreen(); else if (document.webkitExitFullscreen) document.webkitExitFullscreen(); else if (document.msExitFullscreen) document.msExitFullscreen(); } }

        function updateTimeDisplay(current, duration) { const formatTime = (s) => { const m = Math.floor(s / 60); const sec = Math.floor(s % 60); return `${m.toString().padStart(2, '0')}:${sec.toString().padStart(2, '0')}`; }; timeDisplay.textContent = duration > 0 ? `${formatTime(current)} / ${formatTime(duration)}` : '00:00 / 00:00'; }

        function animatePanorama() { requestAnimationFrame(animatePanorama); if (!isGyroEnabled) updateCamera(); if (!currentMediaType && !isGyroEnabled && !isUserInteracting) { lon += autoRotationSpeed; lat = Math.max(-85, Math.min(85, lat)); phi = THREE.MathUtils.degToRad(90 - lat); theta = THREE.MathUtils.degToRad(lon); const x = 500 * Math.sin(phi) * Math.cos(theta); const y = 500 * Math.cos(phi); const z = 500 * Math.sin(phi) * Math.sin(theta); camera.lookAt(x, y, z); } if (video && currentMediaType === 'video' && videoTexture) { videoTexture.needsUpdate = true; } renderer.render(scene, camera); }
